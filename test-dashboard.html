Full pipeline (scam transcript)

$body = @{
  session_id  = "manual-test-$(Get-Random)"
  tenant_id   = "qa-tenant"
  subject_id  = "test-subject"
  audio_pcm_base64 = [Convert]::ToBase64String([byte[]]::new(32000))
  sample_rate_hz = 16000
  channels    = 1
  context_score = 0.3
  transcript  = "Please transfer 50000 rupees and share the OTP now"
} | ConvertTo-Json

$r = Invoke-RestMethod -Uri http://localhost:8080/v1/pipeline/process `
  -Method Post -Body $body -ContentType 'application/json' -TimeoutSec 60

Write-Host "Risk Score  : $($r.risk_assessment.risk_score)"
Write-Host "Final Action: $($r.final_action)"
Write-Host "Content Risk: $($r.content_risk_signal.score) ($($r.content_risk_signal.detail))"
Write-Host "Degraded    : $($r.degraded)"

Risk fusion (direct — try different signal combinations)

$body = @{
  call_session_id = "fusion-test"
  tenant_id = "qa-tenant"
  synthesis_signal    = @{ score = 0.1; confidence = 0.9; available = $true; detail = "low" }
  speaker_match_signal = @{ score = 0.05; confidence = 0.95; available = $true; detail = "low" }
  contextual_signal   = @{ score = 0.1; confidence = 1.0; available = $true; detail = "low" }
  content_risk_signal = @{ score = 0.9; confidence = 0.85; available = $true; detail = "scam" }
} | ConvertTo-Json -Depth 3

Invoke-RestMethod -Uri http://localhost:8000/v1/assess -Method Post `
  -Body $body -ContentType 'application/json' | ConvertTo-Json -Depth 3
